  [0.00,0.00],[0.17,1.45],[0.28,2.67],[0.44,3.80],
  [0.51,4.95],[0.62,6.04],[0.80,7.17],[0.90,7.9167]

var TRAMOS = [[0,.12],[.12,.28],[.28,.44],[.44,.60],[.60,.80]];
var CIERRE_DESDE = 0.92;
function crearMapa(K){
  var n=K.length,x=[],y=[],d=[],m=[],i;
  for(i=0;i<n;i++){x[i]=K[i][0];y[i]=K[i][1];}
  for(i=0;i<n-1;i++) d[i]=(y[i+1]-y[i])/(x[i+1]-x[i]);
  m[0]=d[0]; m[n-1]=d[n-2];
  for(i=1;i<n-1;i++){
    if(d[i-1]*d[i]<=0){m[i]=0;continue;}
    var w1=2*(x[i+1]-x[i])+(x[i]-x[i-1]), w2=(x[i+1]-x[i])+2*(x[i]-x[i-1]);
    m[i]=(w1+w2)/(w1/d[i-1]+w2/d[i]);
  }
  return function(p){
    if(p<=x[0]) return y[0];
    if(p>=x[n-1]) return y[n-1];
    var k=0; while(p>x[k+1]) k++;
    var h=x[k+1]-x[k], s=(p-x[k])/h, s2=s*s, s3=s2*s;
    return (2*s3-3*s2+1)*y[k]+(s3-2*s2+s)*h*m[k]+(-2*s3+3*s2)*y[k+1]+(s3-s2)*h*m[k+1];
  };
}
var mapa = crearMapa(K);

En aplicar(), cambia la línea del objetivo:

El riel (TRAMOS[i][0] + 0.02) sigue funcionando sin cambios. De 0,90 a 1 el video queda en el último cuadro.

- En los tramos comprimidos se saltan cuadros casi idénticos, lo que no se nota.
